{!Person
    name; +String
    birth; #Date
    getCurrentAge; +(): int
}

{!Student
    classes; _List<Class>
    attend; -(class: Course)
    sleep; -()
}

{!Course
    name; +String
    description; +String
    professor1; +professor: Professor
    location; +String
    times; +List<Time>
    prerequisites; +List<Course>
    students; +List<Student>
}

{!Professor
    classes1; +classes: List<Class>
    teach; -teach(class: Course)
}

{!BankAccount
    owner; +String
    balance; +Currency=0
    deposit; +(amount: Currency)
    withdraw; +(amount: Currency)
}

Student > Person
Professor > Person
Course!professor1 > Professor!teach
Course!students > Student!classes
Course!name > Person!name
